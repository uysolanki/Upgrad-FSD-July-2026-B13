create table emp1(
eno int primary key auto_increment,
ename varchar(15) not null,
salary double check(salary>500),
doj date check(doj <= CURRENT_DATE),
passport varchar(10) unique,
workmode varchar(10) check(workmode in('wfo','wfh','hybrid')),
gender varchar(10) check(gender in('m','f')));

-- salary > 5000 platinum
-- salary > 4000 gold
-- salary > 3000 silver
-- salary > 2000 bronze
-- other plastic

select ename,salary,
case
when salary > 5000 then 'Platinum'
when salary > 4000 then 'Gold'
when salary > 3000 then 'Silver'
when salary > 2000 then 'Bronze'
else 'Plastic'
end as 'Category'
from emp1;



select case
when salary > 5000 then 'Platinum'
when salary > 4000 then 'Gold'
when salary > 3000 then 'Silver'
when salary > 2000 then 'Bronze'
else 'Plastic'
end as 'Category', count(*) as 'Number of Employees'
from emp1
group by Category;



-- display the name of employees whose
-- salary is > than the salaries of all emps working in Jaipur

select ename 
from emp1
where salary > ALL (select salary from emp1 where city like 'Jaipur');

-- display the name of employees whose
-- salary is > than the salaries of any single emps working in Jaipur

select ename 
from emp1
where salary > ANY (select salary from emp1 where city like 'Jaipur');


--relationships

create table chiefminister(
cmno int auto_increment,
cmname varchar(20) not null,
gender varchar(10) check(gender in('m','f')),
age int check(age>=50),
primary key(cmno));


create table state(
sno int auto_increment,
sname varchar(20) not null,
capital varchar(10),
cmno int unique,

primary key(sno),
foreign key(cmno) references chiefminister(cmno));
